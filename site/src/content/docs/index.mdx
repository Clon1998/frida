export const requestYaml = `version: 1
name: "Get order"
seq: 2
method: GET
url: "{{baseUrl}}/orders/:orderId?expand=items"
headers:
- name: X-Correlation-Id
  value: "{{$uuid}}"
  enabled: true
auth:
  mode: inherit
scripts:
  post: |
    fri.setVar("lastOrderId", res.body.id);
tests: |
  test("status is 200", () => expect(res.status).to.equal(200));`;
